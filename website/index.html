using Azure.Storage.Blobs;

// Standard local connection string
string connStr = "DefaultEndpointsProtocol=http;AccountName=devstoreaccount1;AccountKey=mock;BlobEndpoint=http://localhost:10000/devstoreaccount1;";

var client = new BlobServiceClient(connStr);
var container = client.GetBlobContainerClient("invoices");
await container.CreateIfNotExistsAsync();

Console.WriteLine("Container created on Floci-Azure in < 1ms!");
provider "azurerm" {
  features {}

  custom_resource_provider_endpoints = {
    "Microsoft.Storage" = "http://localhost:10000"
    "Microsoft.KeyVault"= "http://localhost:10000"

}

resource "azurerm_storage_container" "local" {
  name                 = "tf-local-container"

from azure.storage.blob import BlobServiceClient

conn = "DefaultEndpointsProtocol=http;AccountName=devstoreaccount1;AccountKey=mock;BlobEndpoint=http://localhost:10000/devstoreaccount1;"
client = BlobServiceClient.from_connection_string(conn)
container = client.create_container("reports")
print("Container created locally:", container.container_name)
import { BlobServiceClient, AnonymousCredential } from "@azure/storage-blob";

const blobService = new BlobServiceClient(
  "http://localhost:10000/devstoreaccount1",
  new AnonymousCredential()
);

const container = blobService.getContainerClient("assets");
await container.create();
console.log("Container ready on Floci-Azure!");
